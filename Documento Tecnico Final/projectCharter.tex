	\ESCOMPchItem{Responsable:}{Héctor Alberto Morales Martínez, desarrollador principal}
	\ESCOMPchItem{Autoriza:}{Comisión de Trabajos Terminales (CATT)}
	\ESCOMPchItem{Background/Contexto:}{Miles de NNA quedan en orfandad por feminicidios anualmente; actualmente no existe un sistema centralizado de seguimiento para apoyarlos y visibilizar su situación.}
	\ESCOMPchItem{Beneficios esperados:}{Visibilidad del fenómeno, reducción radical del trabajo manual de monitoreo para ONGs y estructuración de una base de datos histórica mediante PNL.}
	\ESCOMPchItem{Costo estimado:}{Proyecto Académico de Investigación (TT1/TT2)}
	\ESCOMPchSubSubItem{Fecha de inicio:}{Septiembre 2025}{\bf Fecha de término:}{Junio 2026}
	\ESCOMPchItem{Objetivo:}{Desarrollar un sistema automatizado e inteligente de recolección, análisis PNL y visualización de eventos de orfandad por feminicidio en México.}
	\ESCOMPchSec{Entregables Principales}
	\ESCOMPchSubItem{}{SIST-01}{Sistema web orquestado en microservicios contenerizados (Docker Compose)}
	\ESCOMPchSubItem{}{DOC-01}{Documentación Técnica (este documento)}
	\ESCOMPchSubItem{}{COD-01}{Código fuente modular en repositorio}
	\ESCOMPchSec{Alcance del proyecto}
	\ESCOMPchItem{Incluye:}{
		\begin{Titemize}
			\Titem Recolección multi-fuente con mas de 50 feeds RSS y Google News API.
			\Titem Clasificación semántica con el modelo \textbf{BETO Fine-tuned} localmente.
			\Titem Eliminación de ruido con clasificación escalonada.
			\Titem Modelado automático de tópicos con \textbf{BERTopic}.
			\Titem Persistencia con PostgreSQL y dashboard web con modulo de investigación profunda.
		\end{Titemize}
	}
	\ESCOMPchItem{Excluye:}{
		\begin{Titemize}
			\Titem Identificación, almacenamiento o filtrado de datos personales completos y sensibles de NNA.
			\Titem Intervención social, apoyo psicológico directo o canalización jurídica oficial ante autoridades.
			\Titem Validación pericial o legal de la veracidad jurídica de los hechos publicados por los medios.
		\end{Titemize}
	}
	\ESCOMPchItem{Metodología:}{Desarrollo evolutivo por prototipos (P0 a P4).}
