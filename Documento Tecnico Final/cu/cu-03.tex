\begin{UseCase}{CU-03}{Visualización de Clústeres Temáticos}{
	Describe la interacción donde el analista consulta los resultados del agrupamiento semántico (BERTopic) y visualiza las "Fichas de Caso" para los eventos consolidados.
}
	\UCitem{Versión}{\color{Gray}
		1.0
	}\UCitem{Autor}{\color{Gray}
		Tesista de Ingeniería
	}\UCitem{Supervisa}{\color{Gray}
		Director de Trabajo Terminal
	}\UCitem{Actor}{
		Analista de Datos / Trabajador Social
	}\UCitem{Propósito}{\begin{Titemize}
		\Titem Reducir la carga cognitiva consolidando múltiples noticias redundantes en grupos de eventos fácticos unificados.
	\end{Titemize}
	}\UCitem{Entradas}{\begin{Titemize}
		\Titem Parámetros de filtrado por fechas o niveles de relevancia.
		\Titem ID del Clúster seleccionado.
	\end{Titemize}
	}\UCitem{Origen}{\begin{Titemize}
		\Titem Clics del usuario en el menú y tarjetas interactivas del Dashboard.
	\end{Titemize}
	}\UCitem{Salidas}{\begin{Titemize}
		\Titem Interfaz con métricas clave, palabras c-TF-IDF y lista de noticias asociadas.
		\Titem Renderizado de la Ficha Consolidada.
	\end{Titemize}
	}\UCitem{Destino}{\begin{Titemize}
		\Titem Pantalla del navegador del usuario.
	\end{Titemize}
	}\UCitem{Precondiciones}{\begin{Titemize}
		\Titem Debe existir al menos un proceso de ejecución de BERTopic completado en BD (Tabla de clusters poblada).
	\end{Titemize}
	}\UCitem{Postcondiciones}{\begin{Titemize}
		\Titem Ninguna (Operación de solo lectura).
	\end{Titemize}
	}\UCitem{Errores}{\begin{Titemize}
		\Titem {\bf \hypertarget{CU-03.E1}{E1}}: No existen clústeres para el rango de fechas seleccionado \Trayref{CU-03}{A}.
	\end{Titemize}
	}\UCitem{Tipo}{
		Caso de uso primario
	}\UCitem{Observaciones}{
		Los clusters se representan visualmente con código de colores (Verde=NNA confirmado, Rojo=Descartado).
	}
\end{UseCase}

%--------------------------------------
\begin{UCtrayectoria}
	\UCpaso[\UCActor] Accede a la pestaña "Grupos Temáticos" en el menú lateral del Dashboard.
	\UCpaso[\UCsist] Ejecuta una consulta a PostgreSQL extrayendo los nodos centrales de cada clúster de noticias recientes.
	\UCpaso[\UCsist] Verifica que la consulta retorne datos \ErrorRef{CU-03}{E1}{Sin resultados}.
	\UCpaso[\UCsist] Aplica el mapeo de estilos (tarjetas) y presenta el mapa de clústeres en pantalla.
	\UCpaso[\UCActor] Identifica un grupo relevante y hace clic en el botón \IUbutton{Ver Detalles del Caso}.
	\UCpaso[\UCsist] Consulta las noticias hijas pertenecientes al clúster seleccionado.
	\UCpaso[\UCsist] Renderiza en un modal o ventana nueva la información consolidada usando la lógica del \texttt{investigator.py}.
\end{UCtrayectoria}

%--------------------------------------
\begin{UCtrayectoriaA}{CU-03}{A}{No hay datos en el rango de fechas actual}
	\UCpaso Muestra un mensaje en pantalla (Empty State): "No se encontraron clústeres generados para el periodo seleccionado".
	\UCpaso[] El Caso de Uso termina.
\end{UCtrayectoriaA}
